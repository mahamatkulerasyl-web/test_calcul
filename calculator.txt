<script>
    const resultEl = document.getElementById('result');
    const historyEl = document.getElementById('history');
    let expr = '';

    function display() {
        resultEl.textContent = expr
            .replace(/\//g, '÷')
            .replace(/\*/g, '×')
            .replace(/-/g, '−') || '0';
    }

    function popScreen() {
        resultEl.classList.remove('pop');
        void resultEl.offsetWidth;
        resultEl.classList.add('pop');
    }

    function calc() {
        try {
            let safe = expr.replace(/%/g, '/100');
            if (!safe || /[+\-*/.]$/.test(safe)) return;
            let val = Function('"use strict"; return (' + safe + ')')();
            if (!isFinite(val)) throw 0;
            historyEl.textContent = resultEl.textContent + ' =';
            expr = String(Math.round(val * 1e10) / 1e10);
            display();
            popScreen();
        } catch {
            resultEl.textContent = 'Ошибка';
            expr = '';
            setTimeout(display, 900);
        }
    }

    document.querySelectorAll('.btn').forEach(btn => {
        btn.addEventListener('click', () => {
            const action = btn.dataset.action;
            const value = btn.dataset.value;

            if (action === 'clear') {
                expr = '';
                historyEl.textContent = '';
            } else if (action === 'delete') {
                expr = expr.slice(0, -1);
            } else if (action === 'equals') {
                calc();
                return;
            } else {
                expr += value;
            }
            display();
        });
    });

    // Поддержка клавиатуры
    document.addEventListener('keydown', e => {
        const map = { Enter: '=', '=': '=', Backspace: '⌫', Escape: 'C' };
        if (/[0-9.+\-*/%]/.test(e.key)) {
            expr += e.key;
            display();
        } else if (e.key === 'Enter' || e.key === '=') {
            e.preventDefault();
            calc();
        } else if (e.key === 'Backspace') {
            expr = expr.slice(0, -1);
            display();
        } else if (e.key === 'Escape') {
            expr = '';
            historyEl.textContent = '';
            display();
        }
    });
</script>
